# Rust Configuration

## Cargo Configuration

### Cargo.toml Structure
```toml
[package]
name = "my_project"
version = "0.1.0"
edition = "2021"
authors = ["Your Name <your.email@example.com>"]
license = "MIT OR Apache-2.0"
repository = "https://github.com/username/my_project"
description = "A brief description of your project"
keywords = ["keyword1", "keyword2"]
categories = ["category"]

[dependencies]
# External dependencies
serde = { version = "1.0", features = ["derive"] }
tokio = { version = "1.0", features = ["full"] }

# Local dependencies
my_lib = { path = "../my_lib" }

[dev-dependencies]
# Test-only dependencies
criterion = "0.5"
proptest = "1.0"

[build-dependencies]
# Build script dependencies
cc = "1.0"

[features]
default = ["std"]
std = []
async = ["tokio"]
```

### Cargo Config
Create `.cargo/config.toml` in project root:

```toml
# Build configuration
[build]
target = "x86_64-unknown-linux-gnu"
rustflags = ["-C", "target-cpu=native"]

# Registry configuration
[registry]
default = "crates-io"

[source.crates-io]
registry = "https://github.com/rust-lang/crates.io-index"

# Net configuration
[net]
retry = 2
git-fetch-with-cli = true

# Target-specific configuration
[target.x86_64-unknown-linux-gnu]
linker = "x86_64-linux-gnu-gcc"

[target.wasm32-unknown-unknown]
runner = "wasm-bindgen-test-runner"
```

## Rust Compiler Configuration

### rustflags
Set compiler flags in `.cargo/config.toml`:

```toml
[build]
rustflags = [
    "-C", "target-cpu=native",      # Optimize for current CPU
    "-C", "codegen-units=1",        # Better optimization
    "-W", "clippy::all",            # Enable clippy warnings
    "-D", "warnings",               # Treat warnings as errors
]
```

### Environment Variables
```bash
# Set default edition
export RUSTC_BOOTSTRAP=1

# Enable debug info
export RUSTFLAGS="-C debuginfo=2"

# Set target directory
export CARGO_TARGET_DIR="$HOME/.cargo-target"

# Use custom registry
export CARGO_REGISTRIES_CRATES_IO_PROTOCOL="sparse"
```

## Profile Configuration

### Release Profile
```toml
[profile.release]
lto = true              # Link-time optimization
codegen-units = 1       # Better optimization
panic = "abort"         # Smaller binary
strip = true            # Remove debug symbols
opt-level = 3           # Maximum optimization

[profile.release.package."*"]
opt-level = 2           # Slightly less optimization for dependencies
```

### Debug Profile
```toml
[profile.dev]
debug = true            # Include debug info
opt-level = 0           # No optimization
overflow-checks = true  # Enable overflow checks
```

### Test Profile
```toml
[profile.test]
debug = true
opt-level = 1
overflow-checks = true

[profile.bench]
debug = true
opt-level = 3
```

## Cross-Compilation Setup

### Targets
```bash
# Install targets
rustup target add x86_64-unknown-linux-musl
rustup target add x86_64-pc-windows-gnu
rustup target add wasm32-unknown-unknown
rustup target add aarch64-unknown-linux-gnu
```

### Docker Configuration
Create `Dockerfile` for cross-compilation:

```dockerfile
FROM rust:1.70 as builder

# Install cross-compilation tools
RUN apt-get update && apt-get install -y \
    gcc-x86-64-linux-gnu \
    gcc-aarch64-linux-gnu

# Set up cargo config
RUN mkdir -p /usr/local/cargo/config
COPY cargo-config.toml /usr/local/cargo/config/config.toml

WORKDIR /app
COPY . .
RUN cargo build --target x86_64-unknown-linux-gnu --release
```

## Workspace Configuration

### Workspace Setup
```toml
# Cargo.toml (workspace root)
[workspace]
members = [
    "core",
    "cli",
    "web",
    "shared",
]

[workspace.dependencies]
serde = { version = "1.0", features = ["derive"] }
tokio = { version = "1.0", features = ["full"] }
tracing = "0.1"
thiserror = "1.0"

[profile.release]
lto = true
codegen-units = 1
```

### Member Configuration
```toml
# core/Cargo.toml
[package]
name = "my-project-core"
version = "0.1.0"

[dependencies]
serde = { workspace = true }
tokio = { workspace = true }
```

## Verification

### Check Configuration
```bash
# Verify cargo configuration
cargo config get

# Check toolchain
rustup show

# Verify targets
rustup target list --installed

# Test formatting
cargo fmt --check

# Run clippy
cargo clippy -- -D warnings
```

### Build Verification
```bash
# Debug build
cargo build

# Release build
cargo build --release

# Cross-compilation
cargo build --target x86_64-unknown-linux-musl

# Test all targets
cargo test --all-targets
```
